# Flare loads this map after character creation.
# Omit destination coordinates: the generator places the entrance chunk and
# computes its hero_pos. A fixed position can land inside a wall.

[header]
width=1
height=1
hero_pos=0,0
fogofwar=0

[event]
type=event
location=0,0,1,1
activate=on_load
save_game=true

# Let the initial save create this character's maps/ cache before generation.
# Leave enough engine frames to flush save_game and create the per-slot maps/
# directory before the FIRST procedural map is loaded. The former 1ms
# delay provided no margin for game-loop ordering across Flare versions.
[event]
type=event
location=0,0,1,1
activate=on_load
delay=350ms
intermap_random=maps/town/biomes.txt
